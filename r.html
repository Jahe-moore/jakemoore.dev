# Predicting Flight Delays: A Logistic Regression Approach
#
# Question: can we predict the likelihood of a flight delay from its origin
# airport, scheduled departure time, day of the week and flight distance?
# Data: over one million U.S. flights, January-February 2024 (BTS on-time data).
# A flight counts as "delayed" when wheels-off is 15+ minutes after scheduled departure.

library(tidyverse)
library(pROC)
library(caret)
library(broom)

# 1.0 Load and profile ----
flight_data_raw <- read_csv("flight_data_2024.csv")
glimpse(flight_data_raw)

sum(is.na(flight_data_raw))
(sum(is.na(flight_data_raw)) / (nrow(flight_data_raw) * ncol(flight_data_raw))) * 100

# 2.0 Clean ----
# Drop cancelled flights and delays caused by weather or a late inbound aircraft,
# so the model only sees delays the airport and schedule could explain.
airplane_data <- flight_data_raw %>%
  filter(cancelled == 0,
         weather_delay == 0,
         late_aircraft_delay == 0,
         !is.na(dep_time),   # model uses this
         !is.na(taxi_out)    # model uses this
         ) %>%
  select(-air_time, -taxi_in, -late_aircraft_delay, -weather_delay,
         -cancelled, -origin_city_name, -origin_state_nm, -month,
         -day_of_month, -wheels_on, -year)

removed_rows <- nrow(flight_data_raw) - nrow(airplane_data)
paste(removed_rows, "flights removed.")

# 3.0 Build the target ----
# HHMM integers -> real datetimes; a wheels-off time earlier than departure
# means the flight crossed midnight, so add a day.
airplane_data <- airplane_data %>%
  mutate(
    dep_time_str = sprintf("%02d:%02d", dep_time %/% 100, dep_time %% 100),
    wheels_off_str = sprintf("%02d:%02d", wheels_off %/% 100, wheels_off %% 100),
    dep_datetime = mdy_hm(paste(fl_date, dep_time_str)),
    wheels_off_datetime = mdy_hm(paste(fl_date, wheels_off_str)),
    wheels_off_datetime = if_else(wheels_off_datetime < dep_datetime,
                                  wheels_off_datetime + days(1), wheels_off_datetime),
    delay_time = as.numeric(difftime(wheels_off_datetime, dep_datetime, units = "mins")),
    delayed = ifelse(delay_time >= 15, 1, 0),
    day = factor(day_of_week, levels = 1:7,
                 labels = c("Monday", "Tuesday", "Wednesday", "Thursday",
                            "Friday", "Saturday", "Sunday"))
  )

model_summary <- tidy(logit_model) %>%
  filter(term != "(Intercept)") %>%

top_pos <- model_summary %>% arrange(desc(estimate)) %>% slice(1:5)

# What drives Caterpillar sales? Correlation screening, linear vs cubic fits,
# multiple regression, and removing a predictor that causes multicollinearity.
# Data: 39 quarters of Caterpillar sales with eight macro indicators.

library(ggplot2)
library(readr)
library(corrplot)

df <- read.csv("Caterpillar_TimeSeries.csv")

# 1.0 Distribution of the target (sales in millions) ----
summary(df)
summary(df$Caterpillar_Sales)
summary(df$Interest_Rate)

hist(df$Caterpillar_Sales, col = "skyblue",
     xlab = "Sales (in millions)", main = "Distribution of Sales")
boxplot(df$Caterpillar_Sales, col = "orange", main = "Boxplot of Sales")
plot(density(df$Caterpillar_Sales),
     main = "Density Plot of Sales", xlab = "Sales (in millions)")

# 2.0 Correlations ----
cor(df$Caterpillar_Sales, df$Tariff_Index)
# -0.3048: makes sense. Higher tariffs raise costs, which lowers sales.

df_numeric <- df[sapply(df, is.numeric)]   # keep numeric columns only
cor_matrix <- cor(df_numeric)
corrplot(cor_matrix, method = "color", addCoef.col = "black", tl.cex = 0.8)

# Rank every variable by the strength of its correlation with sales
sales_cor <- cor_matrix[, "Caterpillar_Sales"]
sales_df <- data.frame(Variable = names(sales_cor), Correlation = sales_cor)
top_7_corr <- sales_df[order(-abs(sales_df$Correlation)), ][1:7, ]   # target + 6 predictors
top_7_corr
# The directions make sense:
#   Infrastructure_Bill_Score up -> sales up (more funding, more machinery)
#   Residential_Spending up      -> sales up (housing projects need heavy equipment)
#   Tariff_Index up              -> sales down (less trade, higher costs)

# 3.0 Strongest single predictor: linear vs cubic ----
plot(df$Infrastructure_Bill_Score, df$Caterpillar_Sales)
abline(lm(Caterpillar_Sales ~ Infrastructure_Bill_Score, data = df), col = "red")
lines(lowess(df$Infrastructure_Bill_Score, df$Caterpillar_Sales), col = "darkgreen")
# The LOWESS line curves, so a non-linear model may fit better.

summary(model_linear)

model_cubic <- lm(Caterpillar_Sales ~ poly(Infrastructure_Bill_Score, 3, raw = TRUE), data = df)
summary(model_cubic)
# The cubic model has the higher R-squared, so it explains more of the variance in sales.

ggplot(df, aes(x = Infrastructure_Bill_Score, y = Caterpillar_Sales)) +
  geom_point() +
  geom_smooth(method = "lm", formula = y ~ poly(x, 3), color = "blue")

# U.S. college tuition, 2004-05 to 2015-16: tidy a wide spreadsheet, join regions,
# and turn five plain charts into ones that tell the story on their own.
# Data: us_avg_tuition.xlsx (average in-state tuition by state and school year).

library(tidyverse)
library(readxl)

# 1.0 Tidy: one column per year -> one row per state-year ----
us_avg_tuition_raw <- read_excel("us_avg_tuition.xlsx")

state_tuition_long <- us_avg_tuition_raw %>%
  rename_with(tolower) %>%
  pivot_longer(cols = -state, names_to = "school_year", values_to = "avg_cost") %>%
  mutate(
    state = factor(state),
    school_year = as.character(school_year),
    avg_cost = as.numeric(avg_cost)
  )

# Join census regions from R's built-in state data
state_regions <- tibble(state = state.name, region = state.region)
tuition_with_region <- state_tuition_long %>%
  left_join(state_regions, by = "state")

# 2.0 Summaries ----
