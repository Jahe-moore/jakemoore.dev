# Predicting Flight Delays: A Logistic Regression Approach
#
# Question: can we predict the likelihood of a flight delay from its origin
# airport, scheduled departure time, day of the week and flight distance?
# Data: over one million U.S. flights, January-February 2024 (BTS on-time data).
# A flight counts as "delayed" when wheels-off is 15+ minutes after scheduled departure.

library(tidyverse)
library(pROC)
library(caret)
library(broom)

# 1.0 Load and profile ----
flight_data_raw <- read_csv("flight_data_2024.csv")
glimpse(flight_data_raw)

sum(is.na(flight_data_raw))
(sum(is.na(flight_data_raw)) / (nrow(flight_data_raw) * ncol(flight_data_raw))) * 100

# 2.0 Clean ----
# Drop cancelled flights and delays caused by weather or a late inbound aircraft,
# so the model only sees delays the airport and schedule could explain.
airplane_data <- flight_data_raw %>%
  filter(cancelled == 0,
         weather_delay == 0,
         late_aircraft_delay == 0,
         !is.na(dep_time),   # model uses this
         !is.na(taxi_out)    # model uses this
         ) %>%
  select(-air_time, -taxi_in, -late_aircraft_delay, -weather_delay,
         -cancelled, -origin_city_name, -origin_state_nm, -month,
         -day_of_month, -wheels_on, -year)

removed_rows <- nrow(flight_data_raw) - nrow(airplane_data)
paste(removed_rows, "flights removed.")

# 3.0 Build the target ----
# HHMM integers -> real datetimes; a wheels-off time earlier than departure
# means the flight crossed midnight, so add a day.
airplane_data <- airplane_data %>%
  mutate(
    dep_time_str = sprintf("%02d:%02d", dep_time %/% 100, dep_time %% 100),
    wheels_off_str = sprintf("%02d:%02d", wheels_off %/% 100, wheels_off %% 100),
    dep_datetime = mdy_hm(paste(fl_date, dep_time_str)),
    wheels_off_datetime = mdy_hm(paste(fl_date, wheels_off_str)),
    wheels_off_datetime = if_else(wheels_off_datetime < dep_datetime,
                                  wheels_off_datetime + days(1), wheels_off_datetime),
    delay_time = as.numeric(difftime(wheels_off_datetime, dep_datetime, units = "mins")),
    delayed = ifelse(delay_time >= 15, 1, 0),
    day = factor(day_of_week, levels = 1:7,
                 labels = c("Monday", "Tuesday", "Wednesday", "Thursday",
                            "Friday", "Saturday", "Sunday"))
  )

# 4.0 Explore ----
delay_summary <- airplane_data %>%
  count(delayed) %>%
  mutate(prop = n / sum(n))

ggplot(delay_summary, aes(x = "", y = prop, fill = factor(delayed))) +

       x = "Origin Airport", y = "Percentage of Flights Delayed") +
  scale_fill_gradient(low = "darkgreen", high = "darkred") +
  coord_flip() +
  theme_minimal() +

delay_by_day <- airplane_data %>%
  group_by(day) %>%
  summarise(pct_delay = mean(delayed) * 100)

max_delay_by_day <- delay_by_day %>%
  slice_max(pct_delay, n = 1) %>%
  mutate(type = "Highest Delay Day")

min_delay_by_day <- delay_by_day %>%
  slice_min(pct_delay, n = 1) %>%

delay_day_summary <- rbind(max_delay_by_day, min_delay_by_day)

delay_day_summary

# 5.0 Model ----
# Keep the 50 busiest airports as their own levels; group the rest as "Other"
top_50_airports <- airplane_data %>%
  count(origin, sort = TRUE) %>%
  slice_max(n, n = 50) %>%
  pull(origin)

model_data <- airplane_data %>%
  mutate(origin = ifelse(origin %in% top_50_airports, origin, "Other")) %>%
  select(delayed, origin, dep_time, distance, day) %>%
  na.omit()

model_data$origin <- as.factor(model_data$origin)
model_data$day <- as.factor(model_data$day)
model_data$day <- relevel(model_data$day, ref = "Friday")  # Friday = baseline

set.seed(123)
train_index <- createDataPartition(model_data$delayed, p = 0.8, list = FALSE)
train_data <- model_data[train_index, ]
test_data <- model_data[-train_index, ]

logit_model <- glm(delayed ~ ., data = train_data, family = "binomial")

coefs <- coef(logit_model)
sort(abs(coefs), decreasing = TRUE)[1:5]    # strongest effects
sort(abs(coefs), decreasing = FALSE)[1:5]   # weakest effects

# 6.0 Evaluate on the held-out 20% ----
predictions <- predict(logit_model, test_data, type = "response")
predicted_classes <- ifelse(predictions > 0.5, 1, 0)

conf_matrix <- confusionMatrix(factor(predicted_classes), factor(test_data$delayed))
print(conf_matrix)

roc_obj <- roc(test_data$delayed, predictions)
auc_value <- auc(roc_obj)

plot_data <- data.frame(FPR = 1 - roc_obj$specificities, TPR = roc_obj$sensitivities)

ggplot(plot_data, aes(x = FPR, y = TPR)) +
  geom_line(color = "black", size = 1) +
  geom_abline(intercept = 0, slope = 1, linetype = "dashed", color = "darkred") +
  labs(title = paste0("ROC Curve (AUC = ", round(auc_value, 2), ")"),
       x = "Specificity (False Positive Rate)",
       y = "Sensitivity (True Positive Rate)",
       caption = "Source: flight_data_2024.csv") +
  theme_minimal(base_size = 14)

# 7.0 Which factors move the odds of a delay? ----
model_summary <- tidy(logit_model) %>%
  filter(term != "(Intercept)") %>%
  mutate(odds_ratio = exp(estimate),

           grepl("origin", term) ~ "Airport (Origin)",
           grepl("day", term) ~ "Day of Week",
           TRUE ~ "Numeric"))

top_pos <- model_summary %>% arrange(desc(estimate)) %>% slice(1:5)
ggplot(top_pos, aes(x = reorder(term, estimate), y = estimate)) +

  coord_flip() +

       x = "Predictor", y = "Log-Odds Effect",

# What drives Caterpillar sales? Correlation screening, linear vs cubic fits,
# multiple regression, and removing a predictor that causes multicollinearity.
# Data: 39 quarters of Caterpillar sales with eight macro indicators.

library(ggplot2)
library(readr)
library(corrplot)

df <- read.csv("Caterpillar_TimeSeries.csv")

# 1.0 Distribution of the target (sales in millions) ----
summary(df)
summary(df$Caterpillar_Sales)
summary(df$Interest_Rate)

hist(df$Caterpillar_Sales, col = "skyblue",
     xlab = "Sales (in millions)", main = "Distribution of Sales")
boxplot(df$Caterpillar_Sales, col = "orange", main = "Boxplot of Sales")
plot(density(df$Caterpillar_Sales),
     main = "Density Plot of Sales", xlab = "Sales (in millions)")

# 2.0 Correlations ----
cor(df$Caterpillar_Sales, df$Tariff_Index)
# -0.3048: makes sense. Higher tariffs raise costs, which lowers sales.

df_numeric <- df[sapply(df, is.numeric)]   # keep numeric columns only
cor_matrix <- cor(df_numeric)
corrplot(cor_matrix, method = "color", addCoef.col = "black", tl.cex = 0.8)

# Rank every variable by the strength of its correlation with sales
sales_cor <- cor_matrix[, "Caterpillar_Sales"]
sales_df <- data.frame(Variable = names(sales_cor), Correlation = sales_cor)
top_7_corr <- sales_df[order(-abs(sales_df$Correlation)), ][1:7, ]   # target + 6 predictors
top_7_corr
# The directions make sense:
#   Infrastructure_Bill_Score up -> sales up (more funding, more machinery)
#   Residential_Spending up      -> sales up (housing projects need heavy equipment)
#   Tariff_Index up              -> sales down (less trade, higher costs)

# 3.0 Strongest single predictor: linear vs cubic ----
plot(df$Infrastructure_Bill_Score, df$Caterpillar_Sales)
abline(lm(Caterpillar_Sales ~ Infrastructure_Bill_Score, data = df), col = "red")
lines(lowess(df$Infrastructure_Bill_Score, df$Caterpillar_Sales), col = "darkgreen")
# The LOWESS line curves, so a non-linear model may fit better.

model_linear <- lm(Caterpillar_Sales ~ Infrastructure_Bill_Score, data = df)
summary(model_linear)

model_cubic <- lm(Caterpillar_Sales ~ poly(Infrastructure_Bill_Score, 3, raw = TRUE), data = df)
summary(model_cubic)
# The cubic model has the higher R-squared, so it explains more of the variance in sales.

ggplot(df, aes(x = Infrastructure_Bill_Score, y = Caterpillar_Sales)) +
  geom_point() +
  geom_smooth(method = "lm", formula = y ~ poly(x, 3), color = "blue")

# 4.0 Multiple regression with the top six predictors ----
model_multiple <- lm(Caterpillar_Sales ~ Infrastructure_Bill_Score +
  Residential_Spending + Severe_Weather_Index + Tariff_Index +
  Construction_Spending + Interest_Rate,
  data = df)
summary(model_multiple)

# 5.0 Multicollinearity check ----
corrplot(cor_matrix, method = "color", addCoef.col = "black", tl.cex = 0.8)
# Construction_Spending and Residential_Spending correlate at 0.94.
# (Infrastructure_Bill_Score vs sales doesn't count: that's target vs predictor.)
# The overlap inflates standard errors and blurs which variable drives sales.
# Construction_Spending also has p = 0.7430, far above 0.05, so it adds nothing
# unique. Remove it.

model2_multiple <- lm(Caterpillar_Sales ~ Infrastructure_Bill_Score +
  Residential_Spending + Severe_Weather_Index + Tariff_Index +
  Interest_Rate,
  data = df)
summary(model2_multiple)
# Result:
#   Adjusted R-squared rises from 0.9288 to 0.9307 with one fewer variable.
#   Residential_Spending goes from p = 0.0701 (not significant) to p = 7.33e-07.
#   With Construction_Spending gone, multicollinearity is no longer a concern.

# 120 years of Olympic results: join country codes to regions, validate the
# join, rank regions by medals, and compare the leaders with faceted trends.
# Data: 271,116 athlete-event records (1896-2016) and 206 NOC region codes.

library(tidyverse)

olympic_events_raw <- read_csv("olympic_history_athlete_events.csv")
noc_regions <- read_csv("olympic_history_noc_regions.csv")

# 1.0 Join and keep medal winners only ----
olympic_events <- olympic_events_raw %>%
  left_join(noc_regions, by = 'NOC') %>%
  rename_with(tolower) %>%
  filter(!is.na(medal))

# Validate the join before trusting it
sum(is.na(olympic_events$medal)) == 0
sum(is.na(olympic_events$region)) == 0
ncol(olympic_events) == (ncol(olympic_events_raw) + ncol(noc_regions %>% select(-region)))

# 2.0 Which region leads each medal type in the Summer Games? ----
region_medal_count <- olympic_events %>%

  } else {
    result <- df[1:n, ]
  }
  return(result)
}
